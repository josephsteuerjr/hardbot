# Всё, что нужно, чтобы «pytest из корня» стал правдой.
#
#     pip install -r requirements-dev.txt
#     pytest
#
# ЗАЧЕМ ОТДЕЛЬНЫЙ ФАЙЛ. server/requirements.txt — это то, что ставится в образ
# ядра, и туда лишнего класть нельзя: каждая строчка там становится частью
# продакшена. А тесты покрывают не только сервер: пульт (окно на Qt), движок
# (FastAPI), перенос реестров (Excel через COM). На чистой машине без этих
# библиотек сбор тестов падал на 55 файлах ЕЩЁ ДО первой проверки — и человек
# видел красное полотно вместо результата.
#
# Версии закреплены точно: обновление любой из них должно быть осознанным
# решением, а не сюрпризом от свежего релиза в день установки у заказчика.

# --- то, на чём работает сам сервер -------------------------------------------
-r server/requirements.txt

# --- аналитический движок (engine/) -------------------------------------------
# В пакет для ИТ движок не входит, но каталог лежит в репозитории, и его тесты
# запускаются, если fastapi установлен. Без него они честно пропускаются.
-r engine/requirements.txt

# --- сам прогон ---------------------------------------------------------------
pytest==9.1.1

# Часовые пояса. Код считает время визитов в зонах UZ/TJ (+5 ч), а в Windows и
# в slim-образах Linux базы зон в системе нет вовсе: zoneinfo молча не находит
# «Asia/Tashkent» и тесты времени валятся.
tzdata==2026.2

# --- пульт и перенос реестров (тесты окон и Excel) ----------------------------
# Окно пульта и окно переноса реестров. На сервере, где ставят только ядро, Qt
# не нужен: без него эти тесты ПРОПУСКАЮТСЯ (pytest.importorskip), остальные
# идут как обычно. На голом Linux Qt дополнительно требует системную libGL —
# если её нет, импорт не удастся и тесты окон опять же пропустятся, а не упадут.
PySide6==6.11.2

# Чтение книг Excel (.xlsx/.xlsm/.xlsb/.xls) без самого Excel — этим путём
# пульт разбирает реестр перед отправкой на сервер.
# Диапазон, а не точная версия, намеренно: pult/requirements.txt закрепляет
# 0.7.0 для машины аналитика, и жёсткий пин здесь ссорил бы два файла при
# установке в одно окружение. Тесты одинаково зелены на обеих версиях.
python-calamine>=0.7,<0.9

# COM-обвязка Windows: перенос реестров работает с открытыми книгами Excel
# через неё. Маркер обязателен — на Linux этот пакет не существует, и без
# маркера `pip install` упал бы на первой же строке у инженера заказчика.
pywin32==312; sys_platform == "win32"

# xlwings здесь намеренно НЕТ: живой Excel дёргает единственный ручной тест
# (RUN_EXCEL_COM_TEST=1), а остальные проверки подменяют модуль заглушкой.
# Нужен живой Excel — ставится из pult/requirements.txt.
